```java
@Entity
@Table(name = "sys_user")
public class User {
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;
    
    @Column(unique = true, nullable = false, length = 64)
    private String username;
    
    @Column(unique = true, nullable = false, length = 128)
    private String email;
    
    @Column(name = "password_hash", nullable = false, length = 256)
    private String passwordHash;
    
    @Column(length = 256)
    private String avatar;
    
    @Column(nullable = false)
    private Integer status; // 0-禁用, 1-启用
    
    @Column(name = "created_at")
    private LocalDateTime createdAt;
    
    @Column(name = "updated_at")
    private LocalDateTime updatedAt;
    
    // Getters and Setters
}

// 用户服务接口
public interface UserService {
    User register(UserRegisterDTO dto);
    LoginResult login(LoginDTO dto);
    User getCurrentUser();
    User updateUser(Long userId, UserUpdateDTO dto);
    void changePassword(Long userId, PasswordChangeDTO dto);
}

// 用户服务实现
@Service
public class UserServiceImpl implements UserService {
    @Autowired
    private UserRepository userRepository;
    
    @Autowired
    private PasswordEncoder passwordEncoder;
    
    @Autowired
    private JwtTokenProvider tokenProvider;
    
    @Override
    @Transactional
    public User register(UserRegisterDTO dto) {
        // 检查用户名/邮箱是否已存在
        if (userRepository.existsByUsername(dto.getUsername())) {
            throw new BusinessException("用户名已存在");
        }
        if (userRepository.existsByEmail(dto.getEmail())) {
            throw new BusinessException("邮箱已存在");
        }
        
        User user = new User();
        user.setUsername(dto.getUsername());
        user.setEmail(dto.getEmail());
        user.setPasswordHash(passwordEncoder.encode(dto.getPassword()));
        user.setStatus(1);
        user.setCreatedAt(LocalDateTime.now());
        user.setUpdatedAt(LocalDateTime.now());
        
        return userRepository.save(user);
    }
    
    @Override
    public LoginResult login(LoginDTO dto) {
        User user = userRepository.findByUsername(dto.getUsername())
            .orElseThrow(() -> new BusinessException("用户不存在"));
        
        if (!passwordEncoder.matches(dto.getPassword(), user.getPasswordHash())) {
            throw new BusinessException("密码错误");
        }
        
        if (user.getStatus() != 1) {
            throw new BusinessException("账号已被禁用");
        }
        
        String token = tokenProvider.generateToken(user);
        String refreshToken = tokenProvider.generateRefreshToken(user);
        
        return new LoginResult(token, refreshToken, user);
    }
}
```

```java
@Entity
@Table(name = "agent_permission")
public class AgentPermission {
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;
    
    @Column(name = "user_id", nullable = false)
    private Long userId;
    
    @Column(name = "agent_id", nullable = false)
    private Long agentId;
    
    @Column(nullable = false)
    private Integer status; // 0-待审核, 1-已通过, 2-已拒绝
    
    @Column(name = "apply_reason", length = 512)
    private String applyReason;
    
    @Column(name = "reject_reason", length = 512)
    private String rejectReason;
    
    @Column(name = "approved_at")
    private LocalDateTime approvedAt;
    
    @Column(name = "created_at")
    private LocalDateTime createdAt;
    
    // Getters and Setters
}

// 权限服务
public interface PermissionService {
    AgentPermission applyForAgent(Long userId, Long agentId, String reason);
    AgentPermission getPermissionStatus(Long userId, Long agentId);
    List<AgentPermission> getUserPermissions(Long userId);
    void approvePermission(Long permissionId, Long approverId);
    void rejectPermission(Long permissionId, String reason);
    boolean hasPermission(Long userId, Long agentId);
}

@Service
public class PermissionServiceImpl implements PermissionService {
    @Autowired
    private AgentPermissionRepository permissionRepository;
    
    @Override
    @Transactional
    public AgentPermission applyForAgent(Long userId, Long agentId, String reason) {
        // 检查是否已申请
        Optional<AgentPermission> existing = permissionRepository
            .findByUserIdAndAgentId(userId, agentId);
        
        if (existing.isPresent()) {
            AgentPermission perm = existing.get();
            if (perm.getStatus() == 0) {
                throw new BusinessException("申请正在审核中");
            } else if (perm.getStatus() == 1) {
                throw new BusinessException("您已有该Agent权限");
            }
        }
        
        AgentPermission permission = new AgentPermission();
        permission.setUserId(userId);
        permission.setAgentId(agentId);
        permission.setStatus(0);
        permission.setApplyReason(reason);
        permission.setCreatedAt(LocalDateTime.now());
        
        return permissionRepository.save(permission);
    }
    
    @Override
    public boolean hasPermission(Long userId, Long agentId) {
        return permissionRepository
            .existsByUserIdAndAgentIdAndStatus(userId, agentId, 1);
    }
}
```

```java
// 文章服务
public interface ArticleService {
    Article createArticle(Long userId, ArticleCreateDTO dto);
    Article updateArticle(Long userId, Long articleId, ArticleUpdateDTO dto);
    Article getArticle(Long userId, Long articleId);
    Page<Article> listArticles(Long userId, ArticleQuery query);
    void deleteArticle(Long userId, Long articleId);
    Article publishArticle(Long userId, Long articleId, PublishDTO dto);
}

@Service
public class ArticleServiceImpl implements ArticleService {
    
    @Autowired
    private ArticleRepository articleRepository;
    
    @Autowired
    private PublishService publishService;
    
    @Override
    @Transactional
    public Article createArticle(Long userId, ArticleCreateDTO dto) {
        Article article = new Article();
        article.setUserId(userId);
        article.setTitle(dto.getTitle());
        article.setSummary(dto.getSummary());
        article.setContent(dto.getContent());
        article.setCoverImage(dto.getCoverImage());
        article.setTags(String.join(",", dto.getTags()));
        article.setStatus(0); // 草稿
        article.setCreatedAt(LocalDateTime.now());
        article.setUpdatedAt(LocalDateTime.now());
        
        return articleRepository.save(article);
    }
    
    @Override
    @Transactional
    public Article publishArticle(Long userId, Long articleId, PublishDTO dto) {
        Article article = articleRepository.findById(articleId)
            .orElseThrow(() -> new BusinessException("文章不存在"));
        
        if (!article.getUserId().equals(userId)) {
            throw new BusinessException("无权操作");
        }
        
        // 发布到各平台
        List<PublishResult> results = new ArrayList<>();
        for (String platform : dto.getPlatforms()) {
            PublishResult result = publishService.publish(platform, article);
            results.add(result);
        }
        
        // 更新文章状态
        article.setStatus(1); // 已发布
        article.setPublishedAt(LocalDateTime.now());
        article.setPublishLinks(JSON.toJSONString(results));
        article.setUpdatedAt(LocalDateTime.now());
        
        return articleRepository.save(article);
    }
}
```

### 3.4 发布服务设计

#### 3.4.1 发布策略模式

```java
// 发布策略接口
public interface PublishStrategy {
    String getPlatform();
    PublishResult publish(Article article);
    boolean validateConfig(PublishConfig config);
}

// CSDN发布策略
@Component
public class CSDNPublishStrategy implements PublishStrategy {
    
    @Autowired
    private CSDNApiClient csdnClient;
    
    @Override
    public String getPlatform() {
        return "csdn";
    }
    
    @Override
    public PublishResult publish(Article article) {
        try {
            CSDNPostRequest request = convertToCSDNRequest(article);
            CSDNPostResponse response = csdnClient.createPost(request);
            
            return new PublishResult("csdn", response.getUrl(), true, null);
        } catch (Exception e) {
            return new PublishResult("csdn", null, false, e.getMessage());
        }
    }
    
    private CSDNPostRequest convertToCSDNRequest(Article article) {
        CSDNPostRequest request = new CSDNPostRequest();
        request.setTitle(article.getTitle());
        request.setContent(article.getContent());
        request.setSummary(article.getSummary());
        request.setTags(Arrays.asList(article.getTags().split(",")));
        request.setCoverImage(article.getCoverImage());
        return request;
    }
    
    @Override
    public boolean validateConfig(PublishConfig config) {
        return StringUtils.isNotBlank(config.getApiKey()) &&
               StringUtils.isNotBlank(config.getApiSecret());
    }
}

// 发布服务
@Service
public class PublishServiceImpl implements PublishService {
    
    @Autowired
    private List<PublishStrategy> strategies;
    
    @Autowired
    private PublishConfigRepository configRepository;
    
    @Override
    public PublishResult publish(String platform, Article article) {
        PublishStrategy strategy = strategies.stream()
            .filter(s -> s.getPlatform().equals(platform))
            .findFirst()
            .orElseThrow(() -> new BusinessException("不支持的平台: " + platform));
        
        // 获取用户配置
        PublishConfig config = configRepository
            .findByUserIdAndPlatform(article.getUserId(), platform);
        
        if (!strategy.validateConfig(config)) {
            return new PublishResult(platform, null, false, "平台配置不完整");
        }
        
        return strategy.publish(article);
    }
}
```

### 6.2 权限检查流程

```java
@Aspect
@Component
public class AgentPermissionAspect {
    
    @Autowired
    private PermissionService permissionService;
    
    @Around("@annotation(RequireAgentPermission)")
    public Object checkPermission(ProceedingJoinPoint point) throws Throwable {
        // 获取当前用户
        Long userId = SecurityUtils.getCurrentUserId();
        
        // 获取AgentID
        RequireAgentPermission annotation = 
            ((MethodSignature) point.getSignature())
                .getMethod()
                .getAnnotation(RequireAgentPermission.class);
        Long agentId = annotation.value();
        
        // 检查权限
        if (!permissionService.hasPermission(userId, agentId)) {
            throw new NoPermissionException("没有该Agent的使用权限");
        }
        
        return point.proceed();
    }
}
```
